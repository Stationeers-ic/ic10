# VS Code IC10 Extension Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use compose:subagent (recommended) or compose:execute to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Create a VS Code extension with full IC10 language support: syntax highlighting, linting, debugging, autocompletion, and snippets.

**Architecture:** Separate Language Server (LSP) process for linting/completion, Debug Adapter Protocol (DAP) adapter using Ic10Runner from npm package, TextMate grammar for syntax highlighting.

**Tech Stack:** TypeScript, VS Code Extension API, vscode-languageserver, vscode-debugadapter, @stationeers-ic/ic10

## Global Constraints

- Target: VS Code 1.85.0+
- Node.js 18+
- Use npm package `@stationeers-ic/ic10` for all IC10 operations
- Follow VS Code extension best practices
- All code in TypeScript with strict mode

```typescript
// src/lsp/diagnostics.ts
import {
  Diagnostic,
  DiagnosticSeverity,
  Range,
} from "vscode-languageserver";
import { TextDocument } from "vscode-languageserver-textdocument";
import { ValidateIc10Runner } from "@stationeers-ic/ic10";

export function validateDocument(document: TextDocument): Diagnostic[] {
  const diagnostics: Diagnostic[] = [];
  const text = document.getText();

  const errors = ValidateIc10Runner.validate(text);

  for (const error of errors) {
    const line = Math.max(0, (error.line ?? 1) - 1);
    const range: Range = {
      start: { line, character: 0 },
      end: { line, character: Number.MAX_SAFE_INTEGER },
    };

    diagnostics.push({
      severity:
        error.severity === "critical"
          ? DiagnosticSeverity.Error
          : error.severity === "strong"
          ? DiagnosticSeverity.Error
          : DiagnosticSeverity.Warning,
      range,
      message: error.message,
      source: "ic10",
    });
  }

  return diagnostics;
}
```

```typescript
import { CompletionItem, CompletionItemKind } from "vscode-languageserver";
import { instructions } from "@stationeers-ic/ic10";

const INSTRUCTIONS = Object.keys(instructions);

const REGISTERS = Array.from({ length: 18 }, (_, i) => `r${i}`)
  .concat(["sp", "ra"]);

const DEVICES = ["d0", "d1", "d2", "d3", "d4", "d5", "db"];

const CONSTANTS = ["nan", "pi", "e", "true", "false", "null"];

  for (const inst of INSTRUCTIONS) {

      label: inst,
      kind: CompletionItemKind.Keyword,
      detail: "Instruction",
    });
  }

  for (const reg of REGISTERS) {
    items.push({
      label: reg,
      kind: CompletionItemKind.Variable,
      detail: "Register",
    });
  }

  for (const dev of DEVICES) {
    items.push({

      kind: CompletionItemKind.Variable,

  for (const c of CONSTANTS) {
    items.push({
      label: c,
      kind: CompletionItemKind.Constant,
      detail: "Constant",

  return items;
```

```typescript
  createConnection,
  TextDocuments,
  ProposedFeatures,
  InitializeParams,
  InitializeResult,
  TextDocumentSyncKind,
  CompletionParams,
  CompletionItem,
  DiagnosticParams,
  Diagnostic,
} from "vscode-languageserver/node";

import { validateDocument } from "./diagnostics";
import { getCompletions } from "./completion";

const connection = createConnection(ProposedFeatures.all);
const documents = new TextDocuments(TextDocument);

connection.onInitialize((params: InitializeParams): InitializeResult => {
  return {
    capabilities: {
      textDocumentSync: TextDocumentSyncKind.Incremental,
      completionProvider: {
        triggerCharacters: [" "],
      },
      diagnosticProvider: {
        identifier: "ic10-linter",
      },
    },
  };
});

documents.onDidChangeContent((change) => {
  validate(change.document);
});

connection.onCompletion((params: CompletionParams): CompletionItem[] => {
  return getCompletions();
});

connection.languages.diagnostics.on((params: DiagnosticParams): Diagnostic[] => {
  const document = documents.get(params.textDocument.uri);
  if (!document) return [];
  return validateDocument(document);
});

function validate(document: TextDocument): void {
  const diagnostics = validateDocument(document);
  connection.sendDiagnostics({
    uri: document.uri,
    diagnostics,
  });
}

documents.listen(connection);
connection.listen();
```

```typescript
// src/debug/session.ts
import {
  Ic10Runner,
  Chip,
  Housing,
  Network,
  SandboxHousing,
} from "@stationeers-ic/ic10";

export interface DebugState {
  runner: Ic10Runner;
  chip: Chip;
  breakpoints: Set<number>;
  stepping: boolean;
}

export class IC10DebugSession {
  private state: DebugState | null = null;

  async start(program: string): Promise<void> {
    const chip = new Chip();
    chip.setCode(program);

    const network = new Network("data");
    const housing = new SandboxHousing({ network });

    const runner = new Ic10Runner(housing);
    runner.applyChip(chip);

    this.state = {
      runner,
      chip,
      breakpoints: new Set(),
      stepping: false,
    };
  }

  step(): { line: number; registers: Record<string, number> } | null {
    if (!this.state) return null;

    const runner = this.state.runner;
    runner.step();

    return {
      line: runner.getCurrentLine(),
      registers: this.getRegisters(),
    };
  }

  continue(): { line: number; stopped: boolean } | null {
    if (!this.state) return null;

    const runner = this.state.runner;

    while (!runner.isFinished()) {
      runner.step();

      const currentLine = runner.getCurrentLine();
      if (this.state.breakpoints.has(currentLine)) {
        return { line: currentLine, stopped: true };
      }
    }

    return { line: runner.getCurrentLine(), stopped: false };
  }

  setBreakpoint(line: number): void {
    this.state?.breakpoints.add(line);
  }

  removeBreakpoint(line: number): void {
    this.state?.breakpoints.delete(line);
  }

  getRegisters(): Record<string, number> {
    if (!this.state) return {};

    const chip = this.state.chip;
    const regs: Record<string, number> = {};

    for (let i = 0; i < 18; i++) {
      regs[`r${i}`] = chip.registers.get(i) ?? 0;
    }

    return regs;
  }

  getStack(): number[] {
    if (!this.state) return [];

    return this.state.chip.memory.toArray();
  }

  isFinished(): boolean {
    return this.state?.runner.isFinished() ?? true;
  }

  stop(): void {
    this.state = null;
  }
}
```

```typescript
// src/debug/adapter.ts
import {
  DebugSession,
  InitializedEvent,
  StoppedEvent,
  TerminatedEvent,
  Thread,
  StackFrame,
  Source,
  Breakpoint,
} from "vscode-debugadapter";
import { DebugProtocol } from "vscode-debugprotocol";
import { IC10DebugSession } from "./session";
import * as fs from "fs";

export class IC10DebugAdapter extends DebugSession {
  private session: IC10DebugSession = new IC10DebugSession();
  private threadId = 1;

  protected async initializeRequest(
    response: DebugProtocol.InitializeResponse
  ): Promise<void> {
    response.body = {
      supportsSteppingGranularity: false,
      supportsConditionalBreakpoints: true,
      supportsConfigurationDoneRequest: true,
    };

    this.sendResponse(response);
    this.sendEvent(new InitializedEvent());
  }

  protected async launchRequest(
    response: DebugProtocol.LaunchResponse,
    args: DebugProtocol.LaunchRequestArguments
  ): Promise<void> {
    const program = args.program as string;
    const content = fs.readFileSync(program, "utf-8");

    await this.session.start(content);

    this.sendResponse(response);

    if (args.stopOnEntry) {
      this.sendEvent(new StoppedEvent("entry", this.threadId));
    } else {
      this.continueRequest(response as any, {} as any);
    }
  }

  protected async continueRequest(
    response: DebugProtocol.ContinueResponse
  ): Promise<void> {
    const result = this.session.continue();

    if (result?.stopped) {
      this.sendEvent(
        new StoppedEvent("breakpoint", this.threadId)
      );
    } else if (this.session.isFinished()) {
      this.sendEvent(new TerminatedEvent());
    }

    this.sendResponse(response);
  }

  protected async nextRequest(
    response: DebugProtocol.NextResponse
  ): Promise<void> {
    const result = this.session.step();

    if (result) {
      if (this.session.isFinished()) {
        this.sendEvent(new TerminatedEvent());
      } else {
        this.sendEvent(new StoppedEvent("step", this.threadId));
      }
    }

    this.sendResponse(response);
  }

  protected async setBreakPointsRequest(
    response: DebugProtocol.SetBreakpointsResponse,
    args: DebugProtocol.SetBreakpointsArguments
  ): Promise<void> {
    const breakpoints: Breakpoint[] = [];

    this.session.removeAllBreakpoints();

    for (const bp of args.breakpoints ?? []) {
      this.session.setBreakpoint(bp.line);
      breakpoints.push({
        verified: true,
        line: bp.line,
      } as Breakpoint);
    }

    response.body = { breakpoints };
    this.sendResponse(response);
  }

  protected async stackTraceRequest(
    response: DebugProtocol.StackTraceResponse
  ): Promise<void> {
    const line = this.session.getCurrentLine();

    const frame: StackFrame = {
      id: this.threadId,
      name: "IC10",
      line: line,
      column: 0,
      source: new Source("IC10", ""),
    };

    response.body = {
      stackFrames: [frame],
      totalFrames: 1,
    };

    this.sendResponse(response);
  }

  protected scopesRequest(
    response: DebugProtocol.ScopesResponse
  ): void {
    response.body = {
      scopes: [
        {
          name: "Registers",
          variablesReference: 1,
          expensive: false,
        },
        {
          name: "Stack",
          variablesReference: 2,
          expensive: false,
        },
      ],
    };

    this.sendResponse(response);
  }

  protected async variablesRequest(
    response: DebugProtocol.VariablesResponse,
    args: DebugProtocol.VariablesArguments
  ): Promise<void> {
    if (args.variablesReference === 1) {
      const registers = this.session.getRegisters();
      response.body = {
        variables: Object.entries(registers).map(([name, value]) => ({
          name,
          value: String(value),
          type: "number",
          variablesReference: 0,
        })),
      };
    } else if (args.variablesReference === 2) {
      const stack = this.session.getStack();
      response.body = {
        variables: stack.map((value, index) => ({
          name: `[${index}]`,
          value: String(value),
          type: "number",
          variablesReference: 0,
        })),
      };
    }

    this.sendResponse(response);
  }
}
```
